import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid>
  <Card title="Chat" icon="comment">
    Local LLM chat with memory, tool calling, and character companions. Powered by Ollama. No data leaves your network.
  </Card>
  <Card title="Image Generation" icon="pencil">
    ComfyUI + Stable Diffusion XL. txt2img, img2img, face identity, video, background removal, and LoRA support.
  </Card>
  <Card title="Voice" icon="phone">
    Hands-free wakeword detection, Whisper STT, and Kokoro-82M TTS, all running locally with no cloud dependency.
  </Card>
  <Card title="Companions" icon="star">
    A fully customizable, animated AI buddy with its own name, personality, voice, and look. It breathes, reacts, and lip-syncs as it talks, and it's yours, not built to hook you.
  </Card>
  <Card title="Offline Library" icon="open-book">
    Full Wikipedia, medical references, and more via ZIM archives. Readable offline via the built-in kiwix-serve engine.
  </Card>
  <Card title="Maps" icon="rocket">
    Offline maps with routing via MapLibre + pmtiles + GraphHopper. Full-text geocoder, no tile API required.
  </Card>
  <Card title="Home Inventory" icon="list-format">
    AI-powered device tracker. Photo → VLM identify → web lookup. Service logs, warranty alerts, and manual PDF cache.
  </Card>
  <Card title="Home Control" icon="setting">
    Control your smart home by natural language through Home Assistant. "Turn off the office lights", "lock the front door", "set the bedroom to 50%", with per-user, per-room permissions.
  </Card>
  <Card title="YouTube, Privately" icon="youtube">
    A private front-end for YouTube. Search, browse channels, and watch with no Google account and no recommendation engine. Stream is proxied through the server and keeps a personal library.
  </Card>
  <Card title="Podcasts" icon="seti:audio">
    Turn an article, a video, or any source into an AI-generated audio show with chosen voices and offline-rendered intro and outro music.
  </Card>
  <Card title="Music" icon="seti:audio">
    A fully offline music engine (tonal + @tonejs/midi + spessasynth_core) that renders tracks and stingers in the browser from a downloaded soundfont.
  </Card>
  <Card title="Today Dashboard" icon="document">
    News, on-this-day history, sports, weather, jokes, holidays, and local events on one customizable home page, with an ambient daily briefing that flavors your companion.
  </Card>
  <Card title="App Store" icon="puzzle">
    The admin installs or removes any app per household. Apps have pages, extensions add chat abilities, and install-time disclosure shows exactly what each one touches.
  </Card>
  <Card title="Shopping" icon="magnifier">
    Track prices across Amazon, Walmart, Target, and more from one search box. Price-history charts, drop alerts, a deals feed, coupons, and photo/barcode product identification.
  </Card>
</CardGrid>
